Mã nguồn CSS Design Tokens chuẩn (2 Phiên bản Tông màu)

Toàn bộ biến CSS dưới đây đã được nhúng trực tiếp vào SmartFactory.html và Products.html, hỗ trợ chuyển đổi tức thì giữa :root[data-theme="green"] và :root[data-theme="teal"]. 

/* Phiên bản 1 (Mặc định): TÔNG MÀU XANH LÁ (GREEN) */
:root,
:root[data-theme="green"] {
  /* Typography: Inter (UI & Display) + IBM Plex Mono (Technical/Code) */
  --nx-ff-body: "Inter", system-ui, -apple-system, "Segoe UI", Roboto, sans-serif;
  --nx-ff-display: "Inter", system-ui, -apple-system, "Segoe UI", sans-serif;
  --nx-ff-mono: "IBM Plex Mono", ui-monospace, SFMono-Regular, Consolas, monospace;

  /* Brand Palette — Tông màu Xanh lá (Green) */
  --nx-brand: #008B45;               /* Xanh lá chủ đạo (DFS & icon) */
  --nx-brand-alt: #128243;           /* Biến thể xanh lá chủ đạo */
  --nx-brand-secondary: #2E8B57;     /* Xanh lá phụ / chuyển sắc */
  --nx-brand-secondary-alt: #228B22; /* Biến thể xanh lá phụ */
  --nx-brand-hover: #2E8B57;
  --nx-brand-pressed: #128243;
  --nx-brand-subtle: #E6F4EC;
  --nx-brand-gradient: linear-gradient(135deg, #008B45 0%, #2E8B57 100%);
  --nx-on-brand: #FFFFFF;

  /* Core Neutrals — Đen mờ #262626 & Trắng ngà #F5F5F5 */
  --nx-steel-900: #262626;           /* Đen mờ / Xám rất tối (Tên & chức danh) */
  --nx-steel-800: #303030;
  --nx-steel-700: #3D3D3D;
  --nx-surface: #FFFFFF;
  --nx-surface-alt: #F5F5F5;         /* Trắng ngà / Xám nhạt (Nền chính) */
  --nx-text: #262626;
  --nx-text-muted: #5C6661;
  --nx-text-inverse: #FFFFFF;
  --nx-text-brand: #008B45;
}
